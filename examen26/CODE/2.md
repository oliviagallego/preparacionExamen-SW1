## `/app.js`

```js
const express = require("express");
const path = require("path");
const session = require("express-session");

const loginRouter = require("./routes/login");
const restrictedRouter = require("./routes/restricted");
const adminRouter = require("./routes/admin");

const app = express();

app.set("views", path.join(__dirname, "views"));
app.set("view engine", "ejs");

app.use(express.urlencoded({ extended: false }));
app.use(express.static(path.join(__dirname, "public")));

app.use(session({
  secret: "secret",
  resave: false,
  saveUninitialized: true,
}));

// Flash middleware (message/error)
app.use((req, res, next) => {
  const message = req.session.message;
  const error = req.session.error;
  delete req.session.message;
  delete req.session.error;

  res.locals.message = message || "";
  res.locals.error = error || "";
  res.locals.user = req.session.user || null; // global para header.ejs

  next();
});

// Rutas
app.use("/login", loginRouter);
app.use("/restricted", checkLogin, restrictedRouter);
app.use("/admin", checkLogin, /* TODO: requireAdmin */ adminRouter);

// Logout
app.get("/logout", (req, res) => {
  req.session.destroy(() => res.redirect("/login"));
});

function checkLogin(req, res, next) {
  if (req.session.user) return next();
  req.session.error = "Debes iniciar sesión";
  return res.redirect("/login");
}

// TODO (EJERCICIO): middleware requireAdmin

module.exports = app;
```

## `/routes/login.js`

```js
const express = require("express");
const router = express.Router();
const database = require("../database");

router.get("/", (req, res) => {
  res.render("login", { title: "Login" });
});

router.post("/", async (req, res) => {
  const { user, pass } = req.body;

  if (!user || !pass) {
    req.session.error = "Faltan campos";
    return res.redirect("/login");
  }

  const ok = await database.users.isLoginRight(user, pass);

  if (!ok) {
    req.session.error = "Usuario o contraseña incorrectos";
    return res.redirect("/login");
  }

  // TODO (EJERCICIO): guardar en session el usuario REAL, incluyendo role
  // req.session.user = { username: ..., role: ... };

  req.session.message = "Login correcto";
  return res.redirect("/restricted");
});

module.exports = router;
```

En `/admin`:

* Muestra el **total** de usuarios.
* Añade un link a `/admin/users`.

---

## 3) Tabla de usuarios (obligatorio + punto clave de clase)

En `/admin/users`:

* Muestra una **tabla HTML** con: `id`, `username`, `role`.

---

## 4) Crear usuario (obligatorio)

* GET `/admin/users/new`: formulario con `username`, `pass`, `role`.
* POST `/admin/users`: valida:

  * username y pass obligatorios
  * role por defecto `"USER"`
* En `database.users.create`:

  * no permitir username duplicado
  * crear id y guardar en objeto

---

## 5) Socket.IO (entra y es muy típico)

* En server: implementa evento `joinRoom` y `socket.join(room)`.
* En `/public/javascripts/admin_socket.js`: conecta y entra en room `"admin"`.
* Cuando creas usuario (POST), emite a room `"admin"`:

  * evento `"userCreated"` con `{ username, role }`
* En el cliente, al recibir `"userCreated"`, muestra un `alert()` o escribe en consola.
