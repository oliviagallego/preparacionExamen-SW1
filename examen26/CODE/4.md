### `app.js`

```js
const express = require("express");
const path = require("path");
const session = require("express-session");

const indexRouter = require("./routes/index");
const loginRouter = require("./routes/login");
const adminRouter = require("./routes/admin");
const borrowRouter = require("./routes/borrow");

const app = express();

app.set("views", path.join(__dirname, "views"));
app.set("view engine", "ejs");

app.use(express.urlencoded({ extended: false }));
app.use(express.static(path.join(__dirname, "public")));

app.use(session({
  secret: "secret",
  resave: false,
  saveUninitialized: true,
}));

// flash + user globals
app.use((req, res, next) => {
  const message = req.session.message;
  const error = req.session.error;
  delete req.session.message;
  delete req.session.error;

  res.locals.message = message || "";
  res.locals.error = error || "";
  res.locals.user = req.session.user || null;
  next();
});

app.use("/", indexRouter);
app.use("/login", loginRouter);
app.use("/admin", adminRouter);
app.use("/borrow", borrowRouter);

app.get("/logout", (req, res) => {
  req.session.destroy(() => res.redirect("/login"));
});

module.exports = app;
```

```js

const router = express.Router();
const database = require("../database");

router.get("/", (req, res) => {
  res.render("login", { title: "Login" });
});

router.post("/", async (req, res) => {
  const { user, pass } = req.body;

  const ok = await database.users.isLoginRight(user, pass);
  if (!ok) {
    req.session.error = "Login incorrecto";
    return res.redirect("/login");
  }

  // por ahora guarda algo MUY básico
  req.session.user = { username: user };

  req.session.message = "Login OK";
  return res.redirect("/borrow");
});

module.exports = router;
```

# ENUNCIADO (5 APARTADOS)

## 1) Sesión “usuario real” + roles

Ahora mismo al hacer login se guarda `{ username }`.
Tienes que conseguir que, tras login correcto, se guarde:

* `req.session.user = { username, role }` (role real del usuario)

Y si faltan campos en login:

* `req.session.error = "Faltan campos"`
* redirección al login

---

## 2) Middleware de acceso

Implementa **protección por sesión** y **protección por rol**:

* Para poder entrar a **/borrow** hay que estar logeado.
* Para poder entrar a **/admin/** hay que estar logeado **y** ser `role === "ADMIN"`.
* Si no cumple: redirigir con mensaje (error) a `/login` o `/borrow` según toque.

*(No te dicen dónde se hace: tú decides y lo conectas bien.)*

---

## 3) Admin: tabla + buscador (OBJETO → ARRAY)

En `/admin/books`:

* Ya existe buscador por `q`.
* Ahora debes **mostrar una tabla** con columnas: `id`, `title`, `author`, `available`, `acciones`.
* **OJO**: `database.books.data` es un **objeto**, pero la tabla debe recorrer un **array**.
* Si no hay resultados, debe salir: “No hay resultados”.

---

## 4) Admin: crear y borrar libros (POST)

* Crear libro desde el formulario:

  * valida `title` y `author`
  * si falta algo: error + redirigir al formulario
  * si va bien: crear con id autoincremental (`nextId`) y `available: true`
  * mensaje de éxito y redirigir a `/admin/books`
* Borrar:

  * ruta POST `.../:id/delete`
  * si existe: borrar y mensaje “Libro eliminado”
  * si no existe: error “No existe”
  * redirigir a `/admin/books`

---

## 5) Socket.IO (último ejercicio)

Implementa el patrón de rooms + notificaciones:

* El admin se une a la room `"admin"`.
* Los usuarios de borrow se unen a la room `"borrow"`.
* Cuando un admin **crea** un libro:

  * emitir a la room `"admin"` evento `"bookCreated"` con `{ id, title }`
* Cuando un usuario **pide préstamo**:

  * marcar el libro como no disponible (`available=false`)
  * emitir a `"borrow"` evento `"bookBorrowed"` con `{ id, title }`

En el **cliente**:

* Admin: al recibir `bookCreated`, hacer `console.log(...)`
* Borrow: al recibir `bookBorrowed`, hacer `console.log(...)`
