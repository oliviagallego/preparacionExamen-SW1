### `app.js`

```js
const express = require("express");
const path = require("path");
const sessionMiddleware = require("./config/session");

const indexRouter = require("./routes/index");
const loginRouter = require("./routes/login");
const logoutRouter = require("./routes/logout");
const restrictedRouter = require("./routes/restricted");
const adminRouter = require("./routes/admin");

const app = express();

app.set("views", path.join(__dirname, "views"));
app.set("view engine", "ejs");

app.use(express.urlencoded({ extended: false }));
app.use(express.static(path.join(__dirname, "public")));

app.use(sessionMiddleware);

// Flash middleware (mensaje/error de sesión -> res.locals)
app.use((req, res, next) => {
  const message = req.session.message;
  const error = req.session.error;
  delete req.session.message;
  delete req.session.error;

  res.locals.message = message ? `<p class="ok">${message}</p>` : "";
  res.locals.error = error ? `<p class="err">${error}</p>` : "";
  res.locals.me = req.session.user || null;

  next();
});

app.use("/", indexRouter);
app.use("/login", loginRouter);
app.use("/logout", logoutRouter);
app.use("/restricted", restrictedRouter);
app.use("/admin", adminRouter);

app.use((req, res) => res.status(404).send("404"));

module.exports = app;
```

### `routes/login.js`

```js
const express = require("express");
const database = require("../database");
const router = express.Router();

router.get("/", (req, res) => {
  res.render("login", { title: "Login" });
});

router.post("/", (req, res) => {
  const { username, pass } = req.body;

  if (!username || !pass) {
    req.session.error = "Faltan campos";
    return res.redirect("/login");
  }

  const u = database.users.findByUsername(username);
  if (!u || u.pass !== pass) {
    req.session.error = "Credenciales incorrectas";
    return res.redirect("/login");
  }

  // (IMPORTANTE: aquí guardan algo en sesión, pero está incompleto)
  req.session.user = { username: u.username };

  req.session.message = "Login OK";
  return res.redirect("/restricted");
});

module.exports = router;
```

### `sockets.js`

```js
let _io = null;

function initSockets(server) {
  const { Server } = require("socket.io");
  _io = new Server(server);

  _io.on("connection", (socket) => {
    // base: solo para comprobar que funciona
    socket.emit("server:hello", { ok: true });

    socket.on("room:join", (room) => {
      socket.join(String(room));
    });
  });
}

function io() {
  if (!_io) throw new Error("Socket.io no inicializado");
  return _io;
}

module.exports = { initSockets, io };
```

# ENUNCIADO (5 EJERCICIOS)

## 1) Roles + protección por rol

Modifica/añade lo necesario para que:

* En el login se guarde en sesión el usuario **con `username`, `id` y `role`** (el rol viene de `database.users`).
* Crea un middleware **requireAdmin** que solo deje entrar a usuarios con `role === "ADMIN"`.
* Si no es admin:

  * Si no está logeado: `req.session.error = "Debes iniciar sesión"` y redirige a `/login`
  * Si está logeado pero no es admin: `req.session.error = "No tienes permiso"` y redirige a `/restricted`

> (No te dicen dónde meter el middleware: tú decides dónde queda más limpio y reutilizable.)

## 3) Admin users: tabla + buscador + borrar

Implementa:

### A) GET `/admin/users`

* Soporta buscador `?q=ana`
* Filtra por username (case-insensitive)
* Renderiza `admin_users.ejs` con `{ title: "Usuarios", users, q }`
* En la vista:

  * Form GET con input `q` + botón buscar + link limpiar
  * Tabla con id / username / role + columna acciones
  * Si no hay resultados: fila “No hay resultados”

### B) POST `/admin/users/:id/delete`

* Borra por id en `database.users`
* Si se borra: `req.session.message = "Usuario eliminado"`
* Si no existe: `req.session.error = "No existe ese usuario"`
* Redirige a `/admin/users`
* **Extra típico de examen:** no se puede borrar al propio admin logeado (si lo intenta → error “No puedes borrarte a ti mismo”)

* Borra por id, mensaje OK / error igual que users

## 5) Socket.IO (último ejercicio)

Añade/modifica lo necesario para que:

### A) Room de admins

* Cuando un **admin** entra al panel (`/admin` o cualquier ruta admin), el cliente se une a la room `"admins"`.
* Desde cliente: emite `room:join` con `"admins"`.

### B) Notificaciones en tiempo real

* Cuando un admin borra un usuario **o** borra un producto:

  * emite a la room `"admins"` un evento `admin:notice` con:

    ```js
    { type: "user_deleted" | "product_deleted", text: "..." }
    ```
* En el cliente, al recibir `admin:notice`, se muestra en consola y (si quieres) se inserta en un `<div id="live">...</div>`.

### C) Desconectar sockets del usuario borrado (nivel top)

* Cuando se borra un usuario, si ese usuario tiene sockets abiertos, debes **desconectarlos**.
* Pista: crea una room por usuario tipo `"user:<id>"` y haz que cada socket se una a su room al conectarse (si hay sesión / si puedes asociarlo).
* Luego al borrar: `io().to("user:<id>").disconnectSockets(true)`.
